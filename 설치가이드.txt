타스크 큐 db 부재문제

CREATE TABLE `task_queue` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` varchar(64) DEFAULT NULL,
  `status` enum('waiting','processing','done') DEFAULT 'waiting',
  `created_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=4 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

custom_option.toml

#ai 기능 on/off를 추가한 겁니다.
#rag도 on/off 기능이 정상적으로 동작은 하나, 최초 실행 때 켜지는 부분에서는 아직 문제가 있습니다.
#toml 파일을 수정후 streamlit을 다시 껐다 켜주시길 바랍니다.
[app]
ai = "off" #ai on/off 기능
rag = "off" #ragai on/off 기능
history = "on" #민원히스토리 on/off
setting = "on" #설정 on/off
format = "off" #민원 설정 양식 on/off
admin_password = "1234"
[sub]
accept  = "조속히 처리하겠습니다."
particle_accept = "현장확인 후 조속히 처리하겠음."
unaccept = "검토 이후 요청을 이행하기 어려움."
[format]
format = "1. 귀하의 가정에 행복이 가득하시길 바랍니다.\n\n2. 귀하의 민원내용은 [민원요지]에 관한 것으로 이해(또는 판단) 됩니다.\n\n3. 귀하의 질의사항에 대해 검토한 의견은 다음과 같습니다.\n\n가. [답변내용]\n\n4. 귀하의 질문에 만족스러운 답변이 되었기를 바라며, 답변 내용에 대한 추가 설명이 필요한 경우에는 사하구 [부서명]([이름], ☎[전화번호])에게 연락주시면 친절히 안내해 드리도록 하겠습니다.\n아울러 귀하의 민원처리에 대한 만족도 참여를 부탁드립니다. \n감사합니다."
